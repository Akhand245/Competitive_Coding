class Solution {
public:
    int numSubarrayBoundedMax(vector<int>& nums, int left, int right) {
        int strt =0;
        int wc =0;
        int count=0;

        for(int i=0; i<nums.size(); ++i){
            if(nums[i]> right){
                strt= i+1;
                wc =0;
            }else if(nums[i]>=left && nums[i]<= right){
                wc = i-strt+1;
            }
            count+=wc;
        }

        return count;
        
    }
};
